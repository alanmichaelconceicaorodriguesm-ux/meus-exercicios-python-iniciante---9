# Criando a tupla com as frutas.
frutas = ("maçã", "banana", "laranja", "Uva")


# Mostrando cada fruta.
print("Frutas disponíveis:")

for fruta in frutas:
    print(fruta)


# Mostrando a quantidade de elementos.
qtde_frutas = len(frutas)

print(f"Quantidade de frutas: {qtde_frutas}")
